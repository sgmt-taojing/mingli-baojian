// toast
ml.toast.success('已保存');
ml.toast.error('网络异常');

// modal
ml.modal.open('modalBasic');
ml.modal.close('modalBasic', 'api');
const ok = await ml.modal.confirm({ title:'确认删除', message:'此操作不可撤销' });

// tab
ml.tab.setActive('demoTab', 2);
const idx = ml.tab.getActive('demoTab');

// card
const c = ml.card.create({ title:'新卡片', body:'内容', variant:'glass' });

// accordion
ml.accordion.openAll('faqMultiple');
ml.accordion.closeAll('faqMultiple');
